import pandas as pd

cnt = 0

    file = f'fake_data_1_prefix_{idx}.xlsx'
    df = pd.read_excel(file,sheet_name='Sheet1',header=None)
    df.columns = ['姓名','电话号码','身份证号']
    for index,row in df.iterrows():
        if len(row['姓名']) == 2 and index != 0:
            cnt += 1
print(cnt)
